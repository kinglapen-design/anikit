``` css
/* 总容器：标题栏 + 内容区域 */
box-sizing: border-box;

/* Auto layout */
display: flex;
flex-direction: column;

height: 343px; /* 具体由内容而定 */

background: C5;

box-shadow: 0px 4px 16px rgba(0, 0, 0, 0.4); /* 统一悬浮投影，见 [[01_样式 token#07. 阴影]] */

/* 标题栏：标题 + 关闭按钮 */
box-sizing: border-box;

/* Auto layout */
display: flex;
flex-direction: row;

align-items: center;
padding: 8px 8px 8px 16px;
gap: 10px;

width: 450px; /* 具体由总容器而定 */
height: 44px; /* 具体由总容器而定 */

/* Inside auto layout */

flex-grow: 0;

/* 标题 */

/* 浅色（文字，图标） */

/* Inside auto layout */
flex: none;
```
